{# Macro: breadcrumb(items) Renders a DaisyUI breadcrumb bar. items: list of (label, url) tuples. Last item should have url=none to render as current page. Usage: {% from "shared/components/breadcrumb.html" import breadcrumb %} {{ breadcrumb([ ('Admin', url_for('admin.dashboard')), ('Furgoni', none), ]) }} #} {% macro breadcrumb(items) %}
    {% for label, url in items %}
  • {% if url %}{{ label }}{% else %}{{ label }}{% endif %}
    • {% endfor %}
{% endmacro %}
